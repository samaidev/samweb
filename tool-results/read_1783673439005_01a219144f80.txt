    31→// Options controls how the browser window is created.
    32→type Options struct {
    33→        Title      string
    34→        Width      int
    35→        Height     int
    36→        EngineName string
    37→        AgentAddr  string
    38→        AgentToken string
    39→        CDPPort    int
    40→}

    42→// Run starts the wails app and the agent HTTP server. Blocks until the
    43→// window is closed.
    44→func Run(opts Options) error {
    45→        if opts.Title == "" {
    46→                opts.Title = "SamWeb"
    47→        }
    48→        if opts.Width <= 0 {
    49→                opts.Width = 1280
    50→        }
    51→        if opts.Height <= 0 {
    52→                opts.Height = 800
    53→        }
    54→        if opts.AgentAddr == "" {
    55→                opts.AgentAddr = "0.0.0.0:7777"
    56→        }
    57→        if opts.CDPPort == 0 {
    58→                opts.CDPPort = 9222
    59→        }
    60→
    61→        // Enable WebView2's remote debugging port for CDP.
    62→        if os.Getenv("WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS") == "" {
    63→                os.Setenv("WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS",
    64→                        "--remote-debugging-port="+fmt.Sprint(opts.CDPPort))
    65→                log.Printf("[browser] set WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS=%s",
    66→                        os.Getenv("WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS"))
    67→        }
    68→
    69→        // Pick the default search engine.
    70→        engine := search.DefaultEngine
    71→        for _, e := range search.Engines() {
    72→                if e.Name == opts.EngineName {
    73→                        engine = e
    74→                        break
    75→                }
    76→        }
    77→
    78→        // Build the agent backend. This is a WailsBackend that uses wails
    79→        // runtime.ExecJS to drive the browser (replaces webview_go's Eval/Bind).
    80→        backend := NewWailsBackend()
    81→
    82→        // Start the agent HTTP server (port 7777). This serves /agent/* for
    83→        // external programs (Python scripts via SSH tunnel) AND /proxy?url= for
    84→        // the UI's iframe, AND /api/config + /api/resolve for the UI's JS.
    85→        agentSrv := agent.NewServer(opts.AgentAddr, opts.AgentToken, backend)
    86→        go func() {
    87→                if err := agentSrv.ListenAndServe(); err != nil {
    88→                        log.Printf("[browser] agent server error: %v", err)
    89→                }
    90→        }()
    91→
    92→        // Also start a UI HTTP server that serves /proxy, /api/config, /api/resolve,
    93→        // /agent/callback, and reverse-proxies /agent/* to the agent server.
    94→        uiLn, err := net.Listen("tcp", "127.0.0.1:0")
    95→        if err != nil {
    96→                return fmt.Errorf("listen ui: %w", err)
    97→        }
    98→        uiPort := uiLn.Addr().(*net.TCPAddr).Port
    99→        uiSrv := newUIServer(uiPort, engine, opts.AgentAddr)
   100→        // Add the callback handler for JS → Go communication.
   101→        uiSrv.callbackHandler = HandleCallbackHTTP(backend)
   102→        go func() {
   103→                if err := http.Serve(uiLn, uiSrv.handler()); err != nil {
   104→                        log.Printf("[browser] UI server error: %v", err)
   105→                }
   106→        }()
   107→
   108→        // Wait for the UI server to be ready.
   109→        waitForReady(uiPort)
   110→
   111→        log.Printf("[browser] UI server on http://127.0.0.1:%d/", uiPort)
   112→        log.Printf("[browser] agent server on %s", opts.AgentAddr)
   113→
   114→        // Connect to CDP in the background.
   115→        if opts.CDPPort > 0 {
   116→                go func() {
   117→                        var cdpClient *cdp.Client
   118→                        var cdpErr error
   119→                        for i := 0; i < 30; i++ {
   120→                                time.Sleep(500 * time.Millisecond)
   121→                                cdpClient, cdpErr = cdp.ConnectToPage(opts.CDPPort)
   122→                                if cdpErr == nil {
   123→                                        break
   124→                                }
   125→                        }
   126→                        if cdpErr != nil {
   127→                                log.Printf("[browser] CDP connect failed after 15s: %v", cdpErr)
   128→                                return
   129→                        }
   130→                        backend.SetCDPClient(cdpClient)
   131→                        log.Printf("[browser] CDP client connected on port %d", opts.CDPPort)
   132→                }()
   133→        }
   134→
   135→        // Prepare the embedded UI assets.
   136→        uiAssets, err := fs.Sub(uiFS, "ui")
   137→        if err != nil {
   138→                return fmt.Errorf("sub ui fs: %w", err)
   139→        }
   140→
   141→        // Create the wails app context.
   142→        ctx, cancel := context.WithCancel(context.Background())
   143→        defer cancel()
   144→
   145→        // Set the backend's context (needed for runtime.ExecJS).
   146→        backend.SetContext(ctx)
   147→
   148→        // Inject the agent bootstrap JS. This defines window.__samwebAgent
   149→        // and window.__agentCallback, same as the webview_go version.
   150→        // In wails, we use OnDomReady to inject JS after the page loads.
   151→        agentJS := agentBootstrapJS(uiPort)
   152→
   153→        // Run the wails app.
   154→        err = wails.Run(&options.App{
   155→                Title:     opts.Title,
   156→                Width:     opts.Width,
   157→                Height:    opts.Height,
   158→                MinWidth:  400,
   159→                MinHeight: 300,
   160→                AssetServer: &assetserver.Options{
   161→                        Assets: uiAssets,
   162→                        Handler: &uiAssetHandler{
   163→                                uiPort:          uiPort,
   164→                                engine:          engine,
   165→                                agentAddr:       opts.AgentAddr,
   166→                                callbackHandler: HandleCallbackHTTP(backend),
   167→                        },
   168→                },
   169→                OnStartup: func(ctx context.Context) {
   170→                        // Save the context for later use.
   171→                        backend.SetContext(ctx)
   172→                        log.Printf("[browser] wails app started")
   173→                },
   174→                OnDomReady: func(ctx context.Context) {
   175→                        // Inject the agent bootstrap JS.
   176→                        wailsRuntime.WindowExecJS(ctx, agentJS)
   177→                        log.Printf("[browser] agent bootstrap JS injected")
   178→                },
   179→                Bind: []interface{}{
   180→                        backend, // expose backend methods to JS
   181→                },
   182→        })
   183→
   184→        // Shutdown.
   185→        _ = agentSrv.Shutdown(gracefulCtx())
   186→        cancel()
   187→        return err
   188→}

   190→// uiAssetHandler serves dynamic routes (/api/config, /api/resolve, /proxy,
   191→// /agent/callback) alongside the embedded static files. wails calls Handler
   192→// for every request; if it returns nil, wails falls back to serving the
   193→// embedded asset.
   194→type uiAssetHandler struct {
   195→        uiPort          int
   196→        engine          search.Engine
   197→        agentAddr       string
   198→        callbackHandler http.HandlerFunc // handles /agent/callback (JS→Go)
   199→}
   200→
   201→func (h *uiAssetHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
   202→        switch {
   203→        case r.URL.Path == "/agent/callback":
   204→                // JS→Go callback. Must NOT be reverse-proxied to the agent
   205→                // server (the agent server doesn't have this route — the
   206→                // callback is handled in-process by the WailsBackend).
   207→                if h.callbackHandler != nil {
   208→                        h.callbackHandler(w, r)
   209→                } else {
   210→                        http.Error(w, "callback handler not configured", http.StatusServiceUnavailable)
   211→                }
   212→        case r.URL.Path == "/api/config":
   213→                w.Header().Set("Content-Type", "application/json")
   214→                _ = json.NewEncoder(w).Encode(map[string]any{
   215→                        "proxyBase":     fmt.Sprintf("http://127.0.0.1:%d/proxy?url=", h.uiPort),
   216→                        "defaultEngine": h.engine.Name,
   217→                        "engines":       search.Engines(),
   218→                })
   219→        case r.URL.Path == "/api/resolve":
   220→                q := r.URL.Query().Get("q")
   221→                engine := h.engine
   222→                if name := r.URL.Query().Get("engine"); name != "" {
   223→                        for _, e := range search.Engines() {
   224→                                if e.Name == name {
   225→                                        engine = e
   226→                                        break
   227→                                }
   228→                        }
   229→                }
   230→                w.Header().Set("Content-Type", "application/json")
   231→                _ = json.NewEncoder(w).Encode(map[string]any{
   232→                        "url": search.Resolve(q, engine),
   233→                })
   234→        case r.URL.Path == "/proxy":
   235→                target := r.URL.Query().Get("url")
   236→                if target == "" {
   237→                        http.Error(w, "missing url parameter", http.StatusBadRequest)
   238→                        return
   239→                }
   240→                proxy.ServeHTTP(w, r, target)
   241→        case strings.HasPrefix(r.URL.Path, "/agent/"):
   242→                // Reverse proxy /agent/* to the agent server.
   243→                agentTarget := "http://" + h.agentAddr
   244→                agentProxy := httputil.NewSingleHostReverseProxy(mustParseURL(agentTarget))
   245→                originalDirector := agentProxy.Director
   246→                agentProxy.Director = func(req *http.Request) {
   247→                        originalDirector(req)
   248→                        req.Host = h.agentAddr
   249→                }
   250→                agentProxy.ServeHTTP(w, r)
   251→        default:
   252→                // Let wails serve the embedded static file.
   253→                http.FileServer(http.FS(mustSubFS())).ServeHTTP(w, r)
   254→        }
   255→}
   256→
   257→// mustSubFS returns the embedded UI filesystem.
   258→func mustSubFS() fs.FS {
   259→        sub, err := fs.Sub(uiFS, "ui")
   260→        if err != nil {
   261→                log.Fatalf("[browser] cannot sub embed fs: %v", err)
   262→        }
   263→        return sub
   264→}
   265→
   266→// mustParseURL parses a URL string, panicking on error.
   267→func mustParseURL(s string) *url.URL {
   268→        if !strings.HasPrefix(s, "http://") && !strings.HasPrefix(s, "https://") {
   269→                s = "http://" + s
   270→        }
   271→        u, err := url.Parse(s)
   272→        if err != nil {
   273→                log.Fatalf("[browser] invalid agent address %q: %v", s, err)
   274→        }
   275→        return u
   276→}

   278→// uiServer is the HTTP server that serves dynamic routes (/api/config,
   279→// /api/resolve, /proxy, /agent/*) alongside the wails AssetServer.
   280→type uiServer struct {
   281→        port            int
   282→        engine          search.Engine
   283→        agentAddr       string // host:port of the agent HTTP server (for reverse proxy)
   284→        callbackHandler http.HandlerFunc
   285→}
   286→
   287→func newUIServer(port int, engine search.Engine, agentAddr string) *uiServer {
   288→        return &uiServer{port: port, engine: engine, agentAddr: agentAddr}
   289→}
   290→
   291→func (s *uiServer) handler() http.Handler {
   292→        mux := http.NewServeMux()
   293→
   294→        mux.HandleFunc("/ready", func(w http.ResponseWriter, r *http.Request) {
   295→                w.WriteHeader(http.StatusOK)
   296→        })
   297→
   298→        mux.HandleFunc("/api/config", func(w http.ResponseWriter, r *http.Request) {
   299→                w.Header().Set("Content-Type", "application/json")
   300→                _ = json.NewEncoder(w).Encode(map[string]any{
   301→                        "proxyBase":     fmt.Sprintf("http://127.0.0.1:%d/proxy?url=", s.port),
   302→                        "defaultEngine": s.engine.Name,
   303→                        "engines":       search.Engines(),
   304→                })
   305→        })
   306→
   307→        mux.HandleFunc("/api/resolve", func(w http.ResponseWriter, r *http.Request) {
   308→                q := r.URL.Query().Get("q")
   309→                engine := s.engine
   310→                if name := r.URL.Query().Get("engine"); name != "" {
   311→                        for _, e := range search.Engines() {
   312→                                if e.Name == name {
   313→                                        engine = e
   314→                                        break
   315→                                }
   316→                        }
   317→                }
   318→                w.Header().Set("Content-Type", "application/json")
   319→                _ = json.NewEncoder(w).Encode(map[string]any{
   320→                        "url": search.Resolve(q, engine),
   321→                })
   322→        })
   323→
   324→        mux.HandleFunc("/proxy", func(w http.ResponseWriter, r *http.Request) {
   325→                target := r.URL.Query().Get("url")
   326→                if target == "" {
   327→                        http.Error(w, "missing url parameter", http.StatusBadRequest)
   328→                        return
   329→                }
   330→                proxy.ServeHTTP(w, r, target)
   331→        })
   332→
   333→        // /agent/callback — JS → Go callback endpoint
   334→        if s.callbackHandler != nil {
   335→                mux.HandleFunc("/agent/callback", s.callbackHandler)
   336→        }
   337→
   338→        // Reverse-proxy /agent/* to the agent server.
   339→        if s.agentAddr != "" {
   340→                agentTarget := "http://" + s.agentAddr
   341→                agentProxy := httputil.NewSingleHostReverseProxy(mustParseURL(agentTarget))
   342→                originalDirector := agentProxy.Director
   343→                agentProxy.Director = func(req *http.Request) {
   344→                        originalDirector(req)
   345→                        req.Host = s.agentAddr
   346→                }
   347→                mux.Handle("/agent/", agentProxy)
   348→        }
   349→
   350→        return mux
   351→}

   353→// waitForReady polls the UI server's /ready endpoint until it responds.
   354→func waitForReady(port int) {
   355→        url := fmt.Sprintf("http://127.0.0.1:%d/ready", port)
   356→        client := &http.Client{Timeout: 500 * time.Millisecond}
   357→        for i := 0; i < 50; i++ {
   358→                resp, err := client.Get(url)
   359→                if err == nil {
   360→                        _ = resp.Body.Close()
   361→                        return
   362→                }
   363→                time.Sleep(100 * time.Millisecond)
   364→        }
   365→        log.Printf("[browser] WARNING: UI server not ready after 5s")
   366→}

   368→// gracefulCtx returns a context that expires after 5 seconds.
   369→func gracefulCtx() context.Context {
   370→        ctx, _ := context.WithTimeout(context.Background(), 5*time.Second)
   371→        return ctx
   372→}

   374→// agentBootstrapJS is the JS that runs once when the webview initializes.
   375→// It defines window.__samwebAgent, the dispatcher used by the backend.
   376→//
   377→// This is the same as the webview_go version, but uses fetch() to call
   378→// /agent/callback (on the UI server) instead of a native binding.
   379→func agentBootstrapJS(uiPort int) string {
   380→        _ = uiPort
   381→        return `
   382→window.__samwebAgent = (function() {
   383→  var UI_BASE = '';  // same-origin; uiAssetHandler reverse-proxies /agent/*
   384→  var iframe = function() { return document.getElementById('view'); };
   385→  var iwin = function() {
   386→    var f = iframe();
   387→    return f ? f.contentWindow : window;
   388→  };
   389→  var idoc = function() {
   390→    var f = iframe();
   391→    if (!f) return document;
   392→    try { return f.contentDocument; } catch (e) { return null; }
   393→  };
   394→
   395→  function getFrameDoc() {
   396→    var d = idoc();
   397→    if (!d) throw new Error('document is not accessible (cross-origin or not loaded)');
   398→    return d;
   399→  }
   400→
   401→  // dispatch is the single entry point invoked from Go via wails runtime.ExecJS.
   402→  // We use same-origin fetch() to POST the result back to /agent/callback
   403→  // (uiAssetHandler reverse-proxies /agent/* to the agent server, but
   404→  // /agent/callback is handled in-process).
   405→  function dispatch(id, method, params) {
   406→    Promise.resolve().then(function() {
   407→      var fn = methods[method];
   408→      if (!fn) throw new Error('unknown agent method: ' + method);
   409→      return fn(params || {});
   410→    }).then(function(result) {
   411→      fetch(UI_BASE + '/agent/callback', {
   412→        method: 'POST',
   413→        headers: {'Content-Type': 'application/json'},
   414→        body: JSON.stringify({id: id, result: JSON.stringify(result === undefined ? null : result), error: ''})
   415→      }).catch(function(e) {
   416→        console.error('callback fetch error:', e);
   417→      });
   418→    }).catch(function(e) {
   419→      var msg = (e && e.message) ? e.message : String(e);
   420→      fetch(UI_BASE + '/agent/callback', {
   421→        method: 'POST',
   422→        headers: {'Content-Type': 'application/json'},
   423→        body: JSON.stringify({id: id, result: '', error: msg})
   424→      }).catch(function(e2) {
   425→        console.error('callback fetch error:', e2);
   426→      });
   427→    });
   428→  }
   429→
   430→  var methods = {
   431→    navigate: function(p) {
   432→      if (typeof window.navigate === 'function') {
   433→        window.navigate(p.url);
   434→      } else {
   435→        iframe().src = '/proxy?url=' + encodeURIComponent(p.url);
   436→      }
   437→      return { ok: true };
   438→    },
   439→    navigateDirect: function(p) {
   440→      var f = iframe();
   441→      if (f) {
   442→        f.removeAttribute('srcdoc');
   443→        f.src = p.url;
   444→      }
   445→      return { ok: true };
   446→    },
   447→    back: function() {
   448→      if (typeof window.goBack === 'function') { window.goBack(); }
   449→      else { window.history.back(); }
   450→      return { ok: true };
   451→    },
   452→    forward: function() {
   453→      if (typeof window.goForward === 'function') { window.goForward(); }
   454→      else { window.history.forward(); }
   455→      return { ok: true };
   456→    },
   457→    reload: function() {
   458→      if (typeof window.reloadActive === 'function') { window.reloadActive(); }
   459→      else { window.location.reload(); }
   460→      return { ok: true };
   461→    },
   462→    stop: function() {
   463→      try { window.stop(); } catch (e) {}
   464→      return { ok: true };
   465→    },
   466→    click: function(p) {
   467→      var d = getFrameDoc();
   468→      var el;
   469→      if (p.selector) {
   470→        el = d.querySelector(p.selector);
   471→        if (!el) throw new Error('element not found: ' + p.selector);
   472→      } else if (p.x !== undefined && p.y !== undefined) {
   473→        el = d.elementFromPoint(p.x, p.y);
   474→        if (!el) throw new Error('no element at (' + p.x + ',' + p.y + ')');
   475→      } else {
   476→        throw new Error('click requires selector or x,y');
   477→      }
   478→      el.scrollIntoView({block: 'center', inline: 'center'});
   479→      var opts = { bubbles: true, cancelable: true, view: iwin() };
   480→      var btn = (p.button === 'middle') ? 1 : (p.button === 'right') ? 2 : 0;
   481→      el.dispatchEvent(new MouseEvent('mousedown', Object.assign({button: btn}, opts)));
   482→      el.dispatchEvent(new MouseEvent('mouseup',   Object.assign({button: btn}, opts)));
   483→      el.dispatchEvent(new MouseEvent('click',     Object.assign({button: btn}, opts)));
   484→      if (p.double) {
   485→        el.dispatchEvent(new MouseEvent('dblclick', opts));
   486→      }
   487→      return { ok: true, tag: el.tagName.toLowerCase(), text: (el.innerText || '').slice(0, 200) };
   488→    },
   489→    eval: function(p) {
   490→      var d = idoc();
   491→      if (!d) throw new Error('document is not accessible (cross-origin or not loaded)');
   492→      var result = (function() { return eval(p.script); }).call(d.defaultView || window);
   493→      if (result && typeof result.then === 'function') {
   494→        return result.then(function(v) { return { value: v === undefined ? null : v }; });
   495→      }
   496→      return { value: result === undefined ? null : result };
   497→    },
   498→    scroll: function(p) {
   499→      var d = getFrameDoc();
   500→      var w = iwin();
   501→      if (p.x !== undefined && p.y !== undefined) {
   502→        w.scrollTo(p.x, p.y);
   503→      } else if (p.selector) {
   504→        var el = d.querySelector(p.selector);
   505→        if (!el) throw new Error('element not found: ' + p.selector);
   506→        el.scrollIntoView({block: 'start', inline: 'start'});
   507→      } else if (p.direction) {
   508→        var amt = p.amount || 400;
   509→        var dx = 0, dy = 0;
   510→        switch (p.direction) {
   511→          case 'down': dy = amt; break;
   512→          case 'up':   dy = -amt; break;
   513→          case 'right': dx = amt; break;
   514→          case 'left':  dx = -amt; break;
   515→        }
   516→        w.scrollBy(dx, dy);
   517→      } else {
   518→        throw new Error('scroll requires x,y or selector or direction');
   519→      }
   520→      return { ok: true, scrollX: w.scrollX, scrollY: w.scrollY };
   521→    },
   522→    type: function(p) {
   523→      var d = getFrameDoc();
   524→      var el;
   525→      if (p.selector) {
   526→        el = d.querySelector(p.selector);
   527→      } else if (p.x !== undefined && p.y !== undefined) {
   528→        el = d.elementFromPoint(p.x, p.y);
   529→      }
   530→      if (!el) throw new Error('element not found for type');
   531→      var setInputValue = null, setTextareaValue = null;
   532→      try {
   533→        setInputValue = Object.getOwnPropertyDescriptor(
   534→          window.HTMLInputElement.prototype, 'value').set;
   535→      } catch (e) {}
   536→      try {
   537→        setTextareaValue = Object.getOwnPropertyDescriptor(
   538→          window.HTMLTextAreaElement.prototype, 'value').set;
   539→      } catch (e) {}
   540→      function applyValue(target, value) {
   541→        if (target instanceof HTMLTextAreaElement && setTextareaValue) {
   542→          setTextareaValue.call(target, value);
   543→        } else if (setInputValue) {
   544→          setInputValue.call(target, value);
   545→        } else {
   546→          target.value = value;
   547→        }
   548→      }
   549→      if (p.clear) {
   550→        if ('value' in el) applyValue(el, '');
   551→        else el.textContent = '';
   552→      }
   553→      if ('value' in el) {
   554→        applyValue(el, el.value + p.text);
   555→        el.dispatchEvent(new InputEvent('input', {bubbles: true, cancelable: true, inputType: 'insertText', data: p.text}));
   556→        el.dispatchEvent(new Event('change', {bubbles: true}));
   557→      } else {
   558→        el.textContent += p.text;
   559→      }
   560→      return { ok: true };
   561→    },
