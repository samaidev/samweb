     1→// Package cdp provides a minimal Chrome DevTools Protocol client that
     2→// connects to a WebView2 instance's remote debugging port and dispatches
     3→// trusted input events (mouse / touch) that bypass the `event.isTrusted`
     4→// check used by anti-bot systems like Aliyun baxia.
     5→//
     6→// This is the engine-level injection that JS dispatchEvent cannot do:
     7→// CDP's Input.dispatchMouseEvent creates events with isTrusted=true,
     8→// exactly as if a real user moved the mouse.
     9→package cdp

    25→// Client is a CDP-over-WebSocket client connected to a running WebView2
    26→// instance. It is goroutine-safe (a single underlying WebSocket guarded
    27→// by a mutex, since the WS protocol does not allow concurrent writes).
    28→type Client struct {
    29→        mu       sync.Mutex
    30→        conn     *websocket.Conn
    31→        nextID   atomic.Int64
    32→        pending  map[int64]chan cdpResponse
    33→        endpoint string // ws://127.0.0.1:9222/devtools/page/<id>
    34→
    35→        // Network capture state
    36→        captureMu        sync.RWMutex
    37→        capturing        bool
    38→        capturedRequests []CapturedRequest
    39→        // requestMap maps CDP requestId to the index in capturedRequests,
    40→        // so we can update the entry when responseReceived / loadingFinished fires.
    41→        requestMap       map[string]int
    42→}
    43→
    44→type cdpResponse struct {
    45→        result json.RawMessage
    46→        err    error
    47→}

    49→// PageTarget describes a single CDP page target as returned by
    50→// /json on the remote debugging port.
    51→type PageTarget struct {
    52→        ID                   string `json:"id"`
    53→        Title                string `json:"title"`
    54→        URL                  string `json:"url"`
    55→        Type                 string `json:"type"`
    56→        WebSocketDebuggerURL string `json:"webSocketDebuggerUrl"`
    57→}

    59→// FindPageTarget queries http://127.0.0.1:<port>/json and returns the
    60→// first "page" target. Returns nil if none found.
    61→func FindPageTarget(port int) (*PageTarget, error) {
    62→        resp, err := http.Get(fmt.Sprintf("http://127.0.0.1:%d/json", port))
    63→        if err != nil {
    64→                return nil, fmt.Errorf("cdp: cannot reach /json on port %d: %w", port, err)
    65→        }
    66→        defer resp.Body.Close()
    67→        var targets []PageTarget
    68→        if err := json.NewDecoder(resp.Body).Decode(&targets); err != nil {
    69→                return nil, fmt.Errorf("cdp: decode /json response: %w", err)
    70→        }
    71→        for i := range targets {
    72→                if targets[i].Type == "page" && targets[i].WebSocketDebuggerURL != "" {
    73→                        return &targets[i], nil
    74→                }
    75→        }
    76→        return nil, fmt.Errorf("cdp: no page target found on port %d", port)
    77→}
    78→
    79→// Connect dials the CDP WebSocket endpoint and starts a reader goroutine
    80→// that dispatches responses to the waiting callers.
    81→func Connect(wsURL string) (*Client, error) {
    82→        dialer := websocket.Dialer{
    83→                HandshakeTimeout: 10 * time.Second,
    84→        }
    85→        conn, _, err := dialer.Dial(wsURL, nil)
    86→        if err != nil {
    87→                return nil, fmt.Errorf("cdp: dial %s: %w", wsURL, err)
    88→        }
    89→        c := &Client{
    90→                conn:       conn,
    91→                pending:    map[int64]chan cdpResponse{},
    92→                endpoint:   wsURL,
    93→                requestMap: map[string]int{},
    94→        }
    95→        go c.readLoop()
    96→        log.Printf("[cdp] connected to %s", wsURL)
    97→        return c, nil
    98→}

   100→// ConnectToPage is a convenience that finds the first page target on
   101→// the given port and connects to it.
   102→func ConnectToPage(port int) (*Client, error) {
   103→        t, err := FindPageTarget(port)
   104→        if err != nil {
   105→                return nil, err
   106→        }
   107→        return Connect(t.WebSocketDebuggerURL)
   108→}

   110→// FindIframeTarget finds a target of type "iframe" on the given port.
   111→// Returns the first iframe target, or nil if none found.
   112→func FindIframeTarget(port int) (*PageTarget, error) {
   113→        resp, err := http.Get(fmt.Sprintf("http://127.0.0.1:%d/json", port))
   114→        if err != nil {
   115→                return nil, fmt.Errorf("cdp: cannot reach /json on port %d: %w", port, err)
   116→        }
   117→        defer resp.Body.Close()
   118→        var targets []PageTarget
   119→        if err := json.NewDecoder(resp.Body).Decode(&targets); err != nil {
   120→                return nil, fmt.Errorf("cdp: decode /json response: %w", err)
   121→        }
   122→        for i := range targets {
   123→                // iframe targets have type "iframe" in newer Chromium, or
   124→                // type "other" with a URL that looks like a punish/captcha page
   125→                t := targets[i]
   126→                if t.Type == "iframe" && t.WebSocketDebuggerURL != "" {
   127→                        return &t, nil
   128→                }
   129→        }
   130→        // Fallback: look for any target whose URL contains "punish" or
   131→        // "baxia" (the baxia captcha iframe)
   132→        for i := range targets {
   133→                t := targets[i]
   134→                if t.WebSocketDebuggerURL != "" && (strings.Contains(t.URL, "punish") || strings.Contains(t.URL, "baxia")) {
   135→                        return &t, nil
   136→                }
   137→        }
   138→        return nil, fmt.Errorf("cdp: no iframe target found on port %d", port)
   139→}

   141→// ConnectToIframe finds and connects to the baxia iframe's CDP target.
   142→// This is separate from the page-level CDP connection. Events dispatched
   143→// from this connection go directly to the iframe's document, bypassing
   144→// the top-level page's event routing.
   145→func ConnectToIframe(port int) (*Client, error) {
   146→        t, err := FindIframeTarget(port)
   147→        if err != nil {
   148→                return nil, err
   149→        }
   150→        return Connect(t.WebSocketDebuggerURL)
   151→}

   153→// AttachToTarget uses CDP Target.attachToTarget to attach to a specific
   154→// target (e.g. an iframe) by target ID. Returns the session ID needed
   155→// for subsequent commands on that target.
   156→//
   157→// This is the "flat session" approach: we send commands with a
   158→// sessionId field, and CDP routes them to the iframe's context.
   159→func (c *Client) AttachToTarget(targetID string) (string, error) {
   160→        resp, err := c.send("Target.attachToTarget", map[string]interface{}{
   161→                "targetId":  targetID,
   162→                "flatten":   true,
   163→        })
   164→        if err != nil {
   165→                return "", err
   166→        }
   167→        var result struct {
   168→                SessionID string `json:"sessionId"`
   169→        }
   170→        if err := json.Unmarshal(resp, &result); err != nil {
   171→                return "", fmt.Errorf("cdp: decode attachToTarget response: %w", err)
   172→        }
   173→        return result.SessionID, nil
   174→}

   176→// GetTargets returns all targets via CDP Target.getTargets.
   177→func (c *Client) GetTargets() ([]PageTarget, error) {
   178→        resp, err := c.send("Target.getTargets", nil)
   179→        if err != nil {
   180→                return nil, err
   181→        }
   182→        var result struct {
   183→                TargetInfos []struct {
   184→                        TargetID         string `json:"targetId"`
   185→                        Type             string `json:"type"`
   186→                        Title            string `json:"title"`
   187→                        URL              string `json:"url"`
   188→                        Attached         bool   `json:"attached"`
   189→                        OpenerID         string `json:"openerId"`
   190→                        CanAccessOpener  bool   `json:"canAccessOpener"`
   191→                        OpenerFrameID    string `json:"openerFrameId"`
   192→                } `json:"targetInfos"`
   193→        }
   194→        if err := json.Unmarshal(resp, &result); err != nil {
   195→                return nil, fmt.Errorf("cdp: decode getTargets response: %w", err)
   196→        }
   197→        out := make([]PageTarget, len(result.TargetInfos))
   198→        for i, t := range result.TargetInfos {
   199→                out[i] = PageTarget{
   200→                        ID:               t.TargetID,
   201→                        Type:             t.Type,
   202→                        Title:            t.Title,
   203→                        URL:              t.URL,
   204→                        WebSocketDebuggerURL: "", // Target.getTargets doesn't return ws URL
   205→                }
   206→        }
   207→        return out, nil
   208→}

   220→        var result struct {
   221→                Cookies []CDPCookie `json:"cookies"`

   229→// CDPCookie is the CDP representation of a browser cookie (from
   230→// Network.getAllCookies). Field names match the CDP spec.
   231→type CDPCookie struct {
   232→        Name     string  `json:"name"`
   233→        Value    string  `json:"value"`
   234→        Domain   string  `json:"domain"`
   235→        Path     string  `json:"path"`
   236→        Expires  float64 `json:"expires"`   // Unix timestamp in seconds (-1 = session)
   237→        Size     int     `json:"size"`
   238→        HTTPOnly bool    `json:"httpOnly"`
   239→        Secure   bool    `json:"secure"`
   240→        Session  bool    `json:"session"`
   241→        SameSite string  `json:"sameSite"`
   242→        Priority string  `json:"priority"`
   243→}

   245→// SetCookie injects a cookie into the browser's cookie store via CDP
   246→// Network.setCookie. Used by LoadCookiesTrusted to restore a previously
   247→// saved cookie jar.
   248→func (c *Client) SetCookie(cookie CDPCookie) error {
   249→        params := map[string]interface{}{
   250→                "name":     cookie.Name,
   251→                "value":    cookie.Value,
   252→                "domain":   cookie.Domain,
   253→                "path":     cookie.Path,
   254→                "httpOnly": cookie.HTTPOnly,
   255→                "secure":   cookie.Secure,
   256→                "sameSite": cookie.SameSite,
   257→        }
   258→        if cookie.Expires > 0 && !cookie.Session {
   259→                params["expires"] = cookie.Expires
   260→        }
   261→        _, err := c.send("Network.setCookie", params)
   262→        return err
   263→}

   314→// ClearDataForOrigin wipes ALL storage (cookies, localStorage,
   315→// sessionStorage, IndexedDB, cache, service workers) for the given

   324→        if storageTypes == "" {
   325→                storageTypes = "all"
   326→        }
   327→        _, err := c.send("Storage.clearDataForOrigin", map[string]interface{}{
   328→                "origin":       origin,
   329→                "storageTypes": storageTypes,
   330→        })
   331→        return err
   332→}

   334→// DumpLocalStorage reads all localStorage entries for the current page
   335→// via Runtime.evaluate. Returns a map of key→value. Must be called
   336→// when the page is currently loaded on the origin whose localStorage
   337→// you want to dump.
   338→func (c *Client) DumpLocalStorage() (map[string]string, error) {
   339→        script := `(function(){
   340→          var out = {};
   341→          try {
   342→            for (var i = 0; i < localStorage.length; i++) {
   343→              var k = localStorage.key(i);
   344→              out[k] = localStorage.getItem(k) || '';
   345→            }
   346→          } catch(e) {}
   347→          return out;
   348→        })()`
   349→        resp, err := c.send("Runtime.evaluate", map[string]interface{}{
   350→                "expression":    script,
   351→                "returnByValue": true,
   352→        })
   353→        if err != nil {
   354→                return nil, err
   355→        }
   356→        var wrap struct {
   357→                Result struct {
   358→                        Value map[string]string `json:"value"`
   359→                } `json:"result"`
   360→        }
   361→        if err := json.Unmarshal(resp, &wrap); err != nil {
   362→                return nil, fmt.Errorf("decode dump localStorage: %w", err)
   363→        }
   364→        return wrap.Result.Value, nil
   365→}

   367→// RestoreLocalStorage writes the given key→value entries into the
   368→// current page's localStorage via Runtime.evaluate. Must be called
   369→// when the page is currently loaded on the target origin (e.g.
   370→// navigate to https://chat.z.ai first, then call this).
   371→func (c *Client) RestoreLocalStorage(entries map[string]string) error {
   372→        entriesJSON, err := json.Marshal(entries)
   373→        if err != nil {
   374→                return fmt.Errorf("marshal localStorage entries: %w", err)
   375→        }
   376→        script := fmt.Sprintf(`(function(){
   377→          var entries = %s;
   378→          try {
   379→            for (var k in entries) {
   380→              if (entries.hasOwnProperty(k)) {
   381→                localStorage.setItem(k, entries[k]);
   382→              }
   383→            }
   384→          } catch(e) { return false; }
   385→          return true;
   386→        })()`, string(entriesJSON))
   387→        _, err = c.send("Runtime.evaluate", map[string]interface{}{
   388→                "expression":    script,
   389→                "returnByValue": true,
   390→        })
   391→        return err
   392→}

   419→// Returns the base64-encoded PNG data (CDP returns base64, not raw
   420→// bytes). The caller should base64-decode it.
   421→//
   422→// Unlike the JS-level screenshot (which uses SVG foreignObject and
   423→// often fails on complex pages), this captures the actual rendered
   424→// pixels from the WebView2 compositor — what the user sees.
   425→func (c *Client) Screenshot(fullPage bool) ([]byte, error) {
   426→        params := map[string]interface{}{
   427→                "format":      "png",
   428→                "fromSurface": true,
   429→        }
   430→        if fullPage {
   431→                // Capture the full scrollable page, not just the viewport.
   432→                params["clip"] = map[string]interface{}{
   433→                        "x":      0,
   434→                        "y":      0,
   435→                        "width":  1280,
   436→                        "height": 8000,
   437→                        "scale":  1,
   438→                }
   439→        }
   440→        resp, err := c.send("Page.captureScreenshot", params)
   441→        if err != nil {
   442→                return nil, err
   443→        }
   444→        var result struct {
   445→                Data string `json:"data"` // base64-encoded PNG
   446→        }
   447→        if err := json.Unmarshal(resp, &result); err != nil {
   448→                return nil, fmt.Errorf("cdp: decode screenshot response: %w", err)
   449→        }
   450→        if result.Data == "" {
   451→                return nil, fmt.Errorf("cdp: screenshot returned empty data")
   452→        }
   453→        // base64-decode
   454→        decoded, err := base64.StdEncoding.DecodeString(result.Data)
   455→        if err != nil {
   456→                return nil, fmt.Errorf("cdp: base64-decode screenshot: %w", err)
   457→        }
   458→        return decoded, nil
   459→}

   461→// EnableNetwork enables the CDP Network domain and captures all network
   462→// requests with full detail (headers, cookies, response body, timing).
   463→// After calling this, GetCapturedRequests returns a list of all requests
   464→// made by the page.
   465→func (c *Client) EnableNetwork() error {
   466→        // Start capturing before enabling
   467→        c.captureMu.Lock()
   468→        c.capturing = true
   469→        c.capturedRequests = nil
   470→        c.requestMap = map[string]int{}
   471→        c.captureMu.Unlock()
   472→        // Enable Network domain with response bodies
   473→        _, err := c.send("Network.enable", map[string]interface{}{
   474→                "maxTotalBufferSize":    50 * 1024 * 1024,
   475→                "maxResourceBufferSize": 10 * 1024 * 1024,
   476→        })
   477→        return err
   478→}

   542→// parseHeaders converts a CDP headers map (map[string]string or
   543→// map[string][]string) into a sorted slice of NetworkHeader.
   544→func parseHeaders(raw json.RawMessage) []NetworkHeader {
   545→        // CDP sometimes sends headers as map[string]interface{} where values
   546→        // can be string or []string. Try the common map[string]string first.
   547→        var m map[string]string
   548→        if err := json.Unmarshal(raw, &m); err == nil {
   549→                out := make([]NetworkHeader, 0, len(m))
   550→                for k, v := range m {
   551→                        out = append(out, NetworkHeader{Name: k, Value: v})
   552→                }
   553→                return out
   554→        }
   555→        // Fallback: map[string]interface{}
   556→        var mi map[string]interface{}
   557→        if err := json.Unmarshal(raw, &mi); err != nil {
   558→                return nil
   559→        }
   560→        out := make([]NetworkHeader, 0, len(mi))
   561→        for k, v := range mi {
   562→                var vs string
   563→                switch val := v.(type) {
   564→                case string:
   565→                        vs = val
   566→                case []interface{}:
   567→                        // join multiple values with ", " like browsers do
   568→                        parts := make([]string, len(val))
   569→                        for i, p := range val {
   570→                                parts[i], _ = p.(string)
   571→                        }
   572→                        vs = strings.Join(parts, ", ")
   573→                }
   574→                out = append(out, NetworkHeader{Name: k, Value: vs})
   575→        }
   576→        return out
   577→}

   579→// parseCookies extracts cookies from a Cookie header value.
   580→func parseCookies(cookieHeader string) []NetworkCookie {
   581→        if cookieHeader == "" {
   582→                return nil
   583→        }
   584→        parts := strings.Split(cookieHeader, "; ")
   585→        cookies := make([]NetworkCookie, 0, len(parts))
   586→        for _, p := range parts {
   587→                p = strings.TrimSpace(p)
   588→                if p == "" {
   589→                        continue
   590→                }
   591→                idx := strings.Index(p, "=")
   592→                if idx < 0 {
   593→                        continue
   594→                }
   595→                cookies = append(cookies, NetworkCookie{
   596→                        Name:  p[:idx],
   597→                        Value: p[idx+1:],
   598→                })
   599→        }
   600→        return cookies
   601→}

   603→// handleEvent processes CDP events (Network.requestWillBeSent,
   604→// Network.responseReceived, Network.loadingFinished) to capture
   605→// full network traffic including headers, cookies, and response bodies.
   606→func (c *Client) handleEvent(method string, params json.RawMessage) {
   607→        c.captureMu.RLock()
   608→        capturing := c.capturing
   609→        c.captureMu.RUnlock()
   610→        if !capturing {
   611→                return
   612→        }
   613→
   614→        switch method {
   615→        case "Network.requestWillBeSent":
   616→                var ev struct {
   617→                        Request struct {
   618→                                URL      string          `json:"url"`
   619→                                Method   string          `json:"method"`
   620→                                PostData string          `json:"postData"`
   621→                                Headers  json.RawMessage `json:"headers"`
   622→                        } `json:"request"`
   623→                        Type      string  `json:"type"`
   624→                        RequestID string  `json:"requestId"`
   625→                        Timestamp float64 `json:"timestamp"`
   626→                        WallTime float64 `json:"wallTime"`
   627→                }
   628→                if err := json.Unmarshal(params, &ev); err != nil {
   629→                        return
   630→                }
   631→
   632→                req := CapturedRequest{
   633→                        RequestID:    ev.RequestID,
   634→                        URL:          ev.Request.URL,
   635→                        Method:       ev.Request.Method,
   636→                        PostData:     ev.Request.PostData,
   637→                        ResourceType: ev.Type,
   638→                        Timestamp:    ev.Timestamp,
   639→                        WallTime:     ev.WallTime,
   640→                }
   641→                // Parse request headers
   642→                if len(ev.Request.Headers) > 0 {
   643→                        req.RequestHeaders = parseHeaders(ev.Request.Headers)
   644→                        // Extract cookies from the Cookie header
   645→                        for _, h := range req.RequestHeaders {
   646→                                if strings.EqualFold(h.Name, "cookie") {
   647→                                        req.Cookies = parseCookies(h.Value)
   648→                                        break
   649→                                }
   650→                        }
   651→                }
   652→
   653→                c.captureMu.Lock()
   654→                idx := len(c.capturedRequests)
   655→                c.capturedRequests = append(c.capturedRequests, req)
   656→                c.requestMap[ev.RequestID] = idx
   657→                c.captureMu.Unlock()
   658→
   659→        case "Network.responseReceived":
   660→                var ev struct {
   661→                        RequestID  string          `json:"requestId"`
   662→                        Type       string          `json:"type"`
   663→                        Response   struct {
   664→                                Status         int             `json:"status"`
   665→                                StatusText     string          `json:"statusText"`
   666→                                Headers        json.RawMessage `json:"headers"`
   667→                                ContentType    string          `json:"mimeType"`
   668→                                FromServiceWorker bool           `json:"fromServiceWorker"`
   669→                        } `json:"response"`
   670→                        Timestamp  float64 `json:"timestamp"`
   671→                }
   672→                if err := json.Unmarshal(params, &ev); err != nil {
   673→                        return
   674→                }
   675→                c.captureMu.Lock()
   676→                idx, ok := c.requestMap[ev.RequestID]
   677→                if ok && idx < len(c.capturedRequests) {
   678→                        c.capturedRequests[idx].Status = ev.Response.Status
   679→                        c.capturedRequests[idx].StatusText = ev.Response.StatusText
   680→                        c.capturedRequests[idx].ResponseContentType = ev.Response.ContentType
   681→                        if ev.Timestamp > 0 {
   682→                                c.capturedRequests[idx].Duration = ev.Timestamp - c.capturedRequests[idx].Timestamp
   683→                        }
   684→                        if len(ev.Response.Headers) > 0 {
   685→                                c.capturedRequests[idx].ResponseHeaders = parseHeaders(ev.Response.Headers)
   686→                        }
   687→                }
   688→                c.captureMu.Unlock()
   689→
   690→        case "Network.loadingFinished":
   691→                var ev struct {
   692→                        RequestID    string  `json:"requestId"`
   693→                        Timestamp    float64 `json:"timestamp"`
   694→                        EncodedDataLength float64 `json:"encodedDataLength"`
   695→                }
   696→                if err := json.Unmarshal(params, &ev); err != nil {
   697→                        return
   698→                }
   699→                // Asynchronously fetch the response body for interesting resources.
   700→                // We do this in a goroutine to avoid blocking the readLoop.
   701→                c.captureMu.RLock()
   702→                idx, ok := c.requestMap[ev.RequestID]
   703→                if !ok || idx >= len(c.capturedRequests) {
   704→                        c.captureMu.RUnlock()
   705→                        return
   706→                }
   707→                req := &c.capturedRequests[idx]
   708→                // Only fetch body for XHR/Fetch/Document types to avoid
   709→                // downloading large images/scripts.
   710→                shouldFetch := req.ResourceType == "XHR" ||
   711→                        req.ResourceType == "Fetch" ||
   712→                        req.ResourceType == "Document" ||
   713→                        req.ResourceType == "WebSocket" ||
   714→                        req.Status == 0 // retry if status not yet set
   715→                c.captureMu.RUnlock()
   716→
   717→                if ev.EncodedDataLength > 0 {
   718→                        c.captureMu.Lock()
   719→                        if idx < len(c.capturedRequests) {
   720→                                c.capturedRequests[idx].ResponseSize = int64(ev.EncodedDataLength)
   721→                        }
   722→                        c.captureMu.Unlock()
   723→                }
   724→
   725→                // Update duration
   726→                if ev.Timestamp > 0 {
   727→                        c.captureMu.Lock()
   728→                        if idx < len(c.capturedRequests) {
   729→                                c.capturedRequests[idx].Duration = ev.Timestamp - c.capturedRequests[idx].Timestamp
   730→                        }
   731→                        c.captureMu.Unlock()
   732→                }
   733→
   734→                if shouldFetch {
   735→                        go c.fetchResponseBody(ev.RequestID)
   736→                }
   737→        }
   738→}

   740→// fetchResponseBody calls Network.getResponseBody for a finished request
   741→// and stores the result in the captured request entry.
   742→func (c *Client) fetchResponseBody(requestID string) {
   743→        resp, err := c.send("Network.getResponseBody", map[string]interface{}{
   744→                "requestId": requestID,
   745→        })
   746→        if err != nil {
   747→                return // body not available (e.g. cached, redirected)
   748→        }
   749→        var result struct {
   750→                Body          string `json:"body"`
   751→                Base64Encoded bool   `json:"base64Encoded"`
   752→        }
   753→        if err := json.Unmarshal(resp, &result); err != nil {
   754→                return
   755→        }
   756→        c.captureMu.Lock()
   757→        defer c.captureMu.Unlock()
   758→        idx, ok := c.requestMap[requestID]
   759→        if !ok || idx >= len(c.capturedRequests) {
   760→                return
   761→        }
   762→        if result.Base64Encoded {
   763→                // Store raw base64, let the consumer decode
   764→                c.capturedRequests[idx].ResponseBody = "[base64]" + result.Body
   765→        } else {
   766→                // Truncate very large bodies to avoid memory issues
   767→                body := result.Body
   768→                if len(body) > 512*1024 {
   769→                        body = body[:512*1024] + "... [truncated]"
   770→                }
   771→                c.capturedRequests[idx].ResponseBody = body
   772→        }
   773→}
   774→func (c *Client) readLoop() {
   775→        for {
   776→                c.mu.Lock()
   777→                conn := c.conn
   778→                c.mu.Unlock()
   779→                if conn == nil {
   780→                        return
   781→                }
   782→                _, data, err := conn.ReadMessage()
   783→                if err != nil {
   784→                        // Connection closed or errored — fail all pending.
   785→                        c.mu.Lock()
   786→                        for id, ch := range c.pending {
   787→                                ch <- cdpResponse{err: err}
   788→                                delete(c.pending, id)
   789→                        }
   790→                        c.mu.Unlock()
   791→                        return
   792→                }
   793→                var msg struct {
   794→                        ID        int64           `json:"id"`
   795→                        Result    json.RawMessage `json:"result"`
   796→                        Error     *struct {
   797→                                Code    int    `json:"code"`
   798→                                Message string `json:"message"`
   799→                        } `json:"error,omitempty"`
   800→                        Method   string          `json:"method,omitempty"` // for events
   801→                        Params   json.RawMessage `json:"params,omitempty"`
   802→                        SessionID string         `json:"sessionId,omitempty"` // flat session
   803→                }
   804→                if err := json.Unmarshal(data, &msg); err != nil {
   805→                        continue
   806→                }
   807→                if msg.ID == 0 {
   808→                        // Event (not a response). Handle Network events if capturing.
   809→                        if msg.Method != "" {
   810→                                c.handleEvent(msg.Method, msg.Params)
   811→                        }
   812→                        continue
   813→                }
   814→                c.mu.Lock()
   815→                ch, ok := c.pending[msg.ID]
   816→                if ok {
   817→                        delete(c.pending, msg.ID)
   818→                }
   819→                c.mu.Unlock()
   820→                if !ok {
   821→                        continue
   822→                }
   823→                if msg.Error != nil {
   824→                        ch <- cdpResponse{err: fmt.Errorf("cdp error %d: %s", msg.Error.Code, msg.Error.Message)}
   825→                } else {
   826→                        ch <- cdpResponse{result: msg.Result}
   827→                }
   828→        }
   829→}

   831→// send sends a CDP command and waits for the response.
   832→func (c *Client) send(method string, params interface{}) (json.RawMessage, error) {
   833→        id := c.nextID.Add(1)
   834→        ch := make(chan cdpResponse, 1)
   835→        c.mu.Lock()
   836→        if c.conn == nil {
   837→                c.mu.Unlock()
   838→                return nil, fmt.Errorf("cdp: not connected")
   839→        }
   840→        c.pending[id] = ch
   841→        payload := struct {
   842→                ID     int64       `json:"id"`
   843→                Method string      `json:"method"`
   844→                Params interface{} `json:"params,omitempty"`
   845→        }{
   846→                ID:     id,
   847→                Method: method,
   848→                Params: params,
   849→        }
   850→        if err := c.conn.WriteJSON(payload); err != nil {
   851→                delete(c.pending, id)
   852→                c.mu.Unlock()
   853→                return nil, fmt.Errorf("cdp: write: %w", err)
   854→        }
   855→        c.mu.Unlock()
   856→
   857→        select {
   858→        case r := <-ch:
   859→                return r.result, r.err
   860→        case <-time.After(5 * time.Second):
   861→                c.mu.Lock()
   862→                delete(c.pending, id)
   863→                c.mu.Unlock()
   864→                return nil, fmt.Errorf("cdp: timeout waiting for %s response", method)
   865→        }
   866→}

   868→// sendAsync sends a CDP command without waiting for the response. Used
   869→// for high-frequency commands like Input.dispatchMouseEvent during a
   870→// drag, where waiting for each response would make the drag take 10x
   871→// longer than the requested duration.
   872→func (c *Client) sendAsync(method string, params interface{}) error {
   873→        id := c.nextID.Add(1)
   874→        c.mu.Lock()
   875→        defer c.mu.Unlock()
   876→        if c.conn == nil {
   877→                return fmt.Errorf("cdp: not connected")
   878→        }
   879→        // Set a write deadline so a stalled WebSocket doesn't block the
   880→        // drag goroutine forever.
   881→        _ = c.conn.SetWriteDeadline(time.Now().Add(5 * time.Second))
   882→        defer c.conn.SetWriteDeadline(time.Time{}) // reset
   883→        payload := struct {
   884→                ID     int64       `json:"id"`
   885→                Method string      `json:"method"`
   886→                Params interface{} `json:"params,omitempty"`
   887→        }{
   888→                ID:     id,
   889→                Method: method,
   890→                Params: params,
   891→        }
   892→        if err := c.conn.WriteJSON(payload); err != nil {
   893→                return fmt.Errorf("cdp: write: %w", err)
   894→        }
   895→        return nil
   896→}

   898→// sendWithSession sends a CDP command to a specific session (e.g. an
   899→// iframe target attached via Target.attachToTarget). The sessionId
   900→// field routes the command to that target's context.
   901→func (c *Client) sendWithSession(method string, params interface{}, sessionID string) (json.RawMessage, error) {
   902→        id := c.nextID.Add(1)
   903→        ch := make(chan cdpResponse, 1)
   904→        c.mu.Lock()
   905→        if c.conn == nil {
   906→                c.mu.Unlock()
   907→                return nil, fmt.Errorf("cdp: not connected")
   908→        }
   909→        c.pending[id] = ch
   910→        payload := struct {
   911→                ID        int64       `json:"id"`
   912→                Method    string      `json:"method"`
   913→                Params    interface{} `json:"params,omitempty"`
   914→                SessionID string      `json:"sessionId,omitempty"`
   915→        }{
   916→                ID:        id,
   917→                Method:    method,
   918→                Params:    params,
   919→                SessionID: sessionID,
   920→        }
   921→        if err := c.conn.WriteJSON(payload); err != nil {
   922→                delete(c.pending, id)
   923→                c.mu.Unlock()
   924→                return nil, fmt.Errorf("cdp: write: %w", err)
   925→        }
   926→        c.mu.Unlock()
   927→
   928→        select {
   929→        case r := <-ch:
   930→                return r.result, r.err
   931→        case <-time.After(10 * time.Second):
   932→                c.mu.Lock()
   933→                delete(c.pending, id)
   934→                c.mu.Unlock()
   935→                return nil, fmt.Errorf("cdp: timeout waiting for %s (session %s) response", method, sessionID)
   936→        }
   937→}

   939→// DispatchRawSession sends a single CDP mouse event to a specific
   940→// session (e.g. iframe target). This is the key method for baxia:
   941→// by dispatching mouseup from the iframe's own session, the event
   942→// routes directly to the iframe's document — same path as mousedown.
   943→func (c *Client) DispatchRawSession(opts RawMouseOpts, sessionID string) error {
   944→        params := dispatchMouseParams{
   945→                Type:       MouseEventType(opts.Type),
   946→                X:          opts.X,
   947→                Y:          opts.Y,
   948→                Button:     MouseButton(opts.Button),
   949→                Buttons:    opts.Buttons,
   950→                ClickCount: opts.ClickCount,

   954→        }

   956→        return err
